Описать процедуру которая будет запоминать «значение» из выделенной строки табличной части данной формы.

```1c
&НаКлиенте
Процедура ЗапомнитьЗначение(Команда)
	// 1. Проверяем доступность буфера обмена
    Если НЕ СредстваБуфераОбмена.ИспользованиеДоступно() Тогда
        Сообщить("Работа с буфером обмена в вашей системе невозможна.");
        Возврат;
    КонецЕсли;

    // 2. Получаем текущую выделенную строку табличной части
    ТекущаяСтрока = Элементы.Информация.ТекущиеДанные;
    
    Если ТекущаяСтрока = Неопределено Тогда
        Сообщить("Пожалуйста, выделите строку для копирования.");
        Возврат;
    КонецЕсли;

    // 3. Получаем значение из колонки "Значение"
    ЗначениеКолонки = ТекущаяСтрока.Значение;
    СтрокаДляКопирования = Строка(ЗначениеКолонки);
    
    // Убираем концевые пробелы, если это текстовая строка, чтобы не смущать пользователя
    Если ТипЗнч(ЗначениеКолонки) = Тип("Строка") Тогда
        СтрокаДляКопирования = СокрЛП(СтрокаДляКопирования);
    КонецЕсли;

    // 4. Формируем элемент буфера обмена с типом "Текст"
    ЭлементБуфера = Новый ЭлементБуфераОбмена(СтандартныйФорматДанныхБуфераОбмена.Текст, СтрокаДляКопирования);
    МассивЭлементов = Новый Массив;
    МассивЭлементов.Добавить(ЭлементБуфера);

    // 5. Отправляем данные в буфер обмена
    Попытка
        СредстваБуфераОбмена.ПоместитьДанныеАсинх(МассивЭлементов);
        //Сообщить("Значение скопировано в буфер обмена: " + СтрокаДляКопирования);
    Исключение
        Сообщить("Не удалось скопировать данные в буфер обмена.");
    КонецПопытки;
КонецПроцедуры

```
