Описание метода в StepsObject который маскирует автоматизацию

```typescript
import { Page, expect, test, Browser, TestInfo } from '@playwright/test';
import { PageObject } from '../pages/PageObject';

export class StepsObject {
    /* Добавляем маскировку автоматизации */
    static async addMaskingAutomation(page: Page): Promise<void> {
        await page.context().addInitScript(() => {
            Object.defineProperty(navigator, 'webdriver', { get: () => undefined });
            (window as any).chrome = { runtime: {} };
        });
    }
}
```

Использование метода маскировки автоматизации

```typescript
import { test, expect } from '@playwright/test';
import { PageObject } from '../.support/pages/PageObject';
import { StepsObject } from '../.support/steps/StepsObject';
import { Helper } from '../.support/Helper';

test('Главная страница', async ({ page }) => {
    await StepsObject.addMaskingAutomation(page); // маскировка
    
    await page.setViewportSize({ width: 1440, height: 900 });
    await test.step('Загрузка', async () => {
        await page.goto(Helper.URL);
    });
});
```
